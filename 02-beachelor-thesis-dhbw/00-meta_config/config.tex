% --- ANPASSUNG FÜR KAPITELÜBERSCHRIFTEN ---
% Entfernt den vertikalen Abstand vor der Kapitelüberschrift. (liegt an Dokumentenklasse scrreprt, die Teil der KOMA-Script-Familie ist) 
\RedeclareSectionCommand[
  beforeskip=0pt, % Setzt den Abstand VOR der Überschrift auf Null
  afterskip=2\baselineskip % Definiert einen sauberen Abstand NACH der Überschrift
]{chapter}

% --- ANPASSUNG FÜR SECTION- UND SUBSECTION-ÜBERSCHRIFTEN ---
\RedeclareSectionCommand[
  beforeskip=0pt,
  afterskip=0.1\baselineskip % Setzt den Abstand DANACH auf eine halbe Zeilenhöhe
]{section}

\RedeclareSectionCommand[
  beforeskip=0pt,
  afterskip=0.1\baselineskip % Setzt den Abstand DANACH auf eine halbe Zeilenhöhe
]{subsection}

\RedeclareSectionCommand[
  beforeskip=0.2\baselineskip plus 0.2\baselineskip minus 0.2\baselineskip,
  afterskip=-2em % Ein negativer Wert sorgt dafür, dass der Text in derselben Zeile weitergeht
]{paragraph}
