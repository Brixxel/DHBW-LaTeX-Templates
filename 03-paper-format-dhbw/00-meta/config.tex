% === MAKECOVER (Tuned Version) ===
\newcommand{\makecover}{%
    % 1. Header berechnen
    \savebox{\headerContentBox}{%
        % [c] für center alignment (vertikal mittig)
        \begin{minipage}[c]{0.6\textwidth}
            \sffamily\footnotesize \textbf{\gHeaderLeft}
        \end{minipage}%
        \hfill
        % [c] für center alignment
        \begin{minipage}[c]{0.35\textwidth}
            \raggedleft
            % Logo vergrößert auf 1.3cm
            \includegraphics[height=1.3cm]{\gHeaderRightLogo} 
        \end{minipage}
    }%
    % 2. Layout starten
    \twocolumn[%
    \begin{@twocolumnfalse}
        \sffamily 
        \raggedright 

        \noindent\usebox{\headerContentBox}
        \par\vspace{0.8em}

        % Article Type
        {\bfseries\footnotesize\MakeUppercase{\gArticleType}}
        \par\vspace{0.1em}

        % Titel mit reduziertem Zeilenabstand
        {%
            \LARGE\bfseries
            \setlength{\baselineskip}{1.1em}  
            \gMyTitle \par
        }
        \vspace{0.5em}  

        % Autoren
        {\large \gAuthorsText \par}
        \vspace{0.5em}

        % Affiliation
        {\footnotesize \gAffiliation \par}
        \vspace{0.2em}

        % Korrespondenz
        {\footnotesize \textbf{Author for correspondence:} \gCorrespondence \par}
        
        \vspace{1.25em}
        
        % Abstract
        \ifx\gMyAbstract\empty
        \else
            \begin{quote}
            \normalfont 
            \small
            \textbf{\sffamily Abstract.} \gMyAbstract
            \end{quote}
            \vspace{1em}
        \fi
        
    \end{@twocolumnfalse}
    ]
}
