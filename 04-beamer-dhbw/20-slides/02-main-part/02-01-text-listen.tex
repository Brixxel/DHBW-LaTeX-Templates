\begin{frame}{Text und Hervorhebungen}
  Standardtext liest sich am besten in \textbf{ruhigen Farben}. Hebe nur das
  hervor, was wirklich wichtig ist – z.\,B. mit \highlight{Akzentfarbe} oder
  \alert{alert-Stil}.

  \vspace{0.8em}
  \textbf{Geschachtelte Liste:}
  \begin{itemize}
    \item Erste Ebene mit rotem Bullet
    \begin{itemize}
      \item Zweite Ebene mit Strich
      \item Noch ein Unterpunkt
    \end{itemize}
    \item Zweiter Hauptpunkt
  \end{itemize}

  \vspace{0.4em}
  \textbf{Nummerierte Liste:}
  \begin{enumerate}
    \item Schritt eins
    \item Schritt zwei
    \item Schritt drei
  \end{enumerate}
\end{frame}
