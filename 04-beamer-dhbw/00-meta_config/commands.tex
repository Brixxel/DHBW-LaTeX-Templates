% Zitatfolie
\newcommand{\quoteslide}[2]{%
  \begin{frame}[plain]
    \vfill
    \begin{center}
      \large\itshape ``#1''\\[1em]
      \normalfont\normalsize --- #2
    \end{center}
    \vfill
  \end{frame}%
}

    \end{tikzpicture}
  \end{frame}%
}
